cmake_minimum_required(VERSION 3.24)
project(katla_window_native LANGUAGES C)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_AUDIO OFF CACHE BOOL "" FORCE)
set(SDL_GPU OFF CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_JOYSTICK OFF CACHE BOOL "" FORCE)
set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
set(SDL_OPENGLES OFF CACHE BOOL "" FORCE)
set(SDL_VULKAN ON CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(SDL_X11 ON CACHE BOOL "" FORCE)
  set(SDL_WAYLAND ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${KATLA_SDL_SOURCE}" sdl)
add_library(katla_window_native SHARED bridge.c)
target_link_libraries(katla_window_native PRIVATE SDL3::SDL3)
target_compile_features(katla_window_native PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(katla_window_native PRIVATE -Wall -Wextra -Werror)
endif()
set_target_properties(katla_window_native PROPERTIES BUILD_RPATH "$ORIGIN;@loader_path" INSTALL_RPATH "$ORIGIN;@loader_path")
install(TARGETS katla_window_native RUNTIME DESTINATION . LIBRARY DESTINATION .)
option(KATLA_BUILD_SMOKE "Build the explicit native window acceptance executable" OFF)
if(KATLA_BUILD_SMOKE)
  add_executable(katla_window_smoke smoke.c)
  target_link_libraries(katla_window_smoke PRIVATE katla_window_native SDL3::SDL3)
  target_compile_features(katla_window_smoke PRIVATE c_std_11)
  set_target_properties(katla_window_smoke PROPERTIES BUILD_RPATH "$ORIGIN;@loader_path" INSTALL_RPATH "$ORIGIN;@loader_path")
  install(TARGETS katla_window_smoke RUNTIME DESTINATION .)
endif()
